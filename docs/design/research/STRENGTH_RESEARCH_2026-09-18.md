Read as:

(v) The one small-compute ablation that isolates augmentation — Tablut (§Q9), C4 rotations on 2 H200s — bundles it with a 2× replay window and reports the final Elo only in a figure.

**Reading it for our regime.** (i) "Board size" means different things across these sources: for KataGo and SAZ it is the edge of a bounded board (a different game at each size, with a CNN built to pool across sizes); for strix and us it is the legal-move HALO around the stones on an unbounded board — the graph encoding is size-agnostic (registry: "Action space is unchanged"), so a radius curriculum changes the branching factor and the game length, not the net. Our own r6 → r8 move was made for ecosystem reasons (registry.toml:76-80: "at radius 6 the encoder refused 3 023 of 8 698 corpus games (34.76%)"), and F-08 found that "Legal-move radius compression at bootstrap" did not move bootstrap quality on the dense lineage ("Median plies identical across all radii") — a bootstrap finding, not a self-play one, cited for scope. (ii) The sources whose game is nearest (SAZ's Gomoku, Soemers' Connect6 — our rules on a bounded board) both report small→large transfer WORKS, and both confound it with budget (Soemers says so). (iii) The one same-game data point, strix, staged win-length AND radius (4/r2 → 6/r8) and found the radius-2 six-in-a-row stage degenerate ("both sides converge to forced draws and the network unlearns win-finding skills"); its checkpoint that beats us is a radius-6-stage net. (iv) The regime argument for a curriculum is throughput: shorter games at small radius mean more games per hour on a serving-bound box (F-47) — SAZ's "order of magnitude less time" is exactly that — but no source separates the throughput gain from a transfer gain, and MiniZero's progressive SIMULATION schedule (the same idea on a different axis) was "weaker … before 54,000 training steps" and lost on Atari.

use mantis_graph::{build_axis_graph, AxisGraph, BuildParams, StoneList, EDGE_FEAT_DIM, NODE_FEAT_DIM};
use std::collections::{BTreeMap, BTreeSet};

const R8: BuildParams = BuildParams { win_length: 6, radius: 8, current_player: 1, moves_remaining: 2, trunk_size: 19 };

// Same 12 elements as mantis_selfplay::replay::sym::rotate_axial (reflect-then-rotate).
fn rotate_axial(q: i32, r: i32, s: usize) -> (i32, i32) {
    let (mut q, mut r) = if s >= 6 { (r, q) } else { (q, r) };
    for _ in 0..(s % 6) { let nq = -r; let nr = q + r; q = nq; r = nr; }
    (q, r)
}
fn hexd(a: (i32, i32), b: (i32, i32)) -> i32 {
    let dq = (b.0 - a.0).abs(); let dr = (b.1 - a.1).abs(); let ds = (b.0 + b.1 - a.0 - a.1).abs();
    dq.max(dr).max(ds)
}
struct Lcg(u64);
impl Lcg {
    fn next(&mut self) -> u64 { self.0 = self.0.wrapping_mul(6364136223846793005).wrapping_add(1442695040888963407); self.0 >> 33 }
    fn range(&mut self, lo: i64, hi: i64) -> i64 { lo + (self.next() as i64 % (hi - lo)) }
}
fn coords(g: &AxisGraph) -> Vec<(i32, i32)> { (0..g.num_nodes()).map(|i| (g.node_coords[2*i], g.node_coords[2*i+1])).collect() }
fn stone_set(g: &AxisGraph) -> BTreeSet<(i32,i32,i8)> {
    let c = coords(g);
    (0..g.n_stones as usize).map(|i| { let own = g.node_feat.0[i*NODE_FEAT_DIM] == 1.0; (c[i].0, c[i].1, if own {1} else {-1}) }).collect()
}
fn legal_set(g: &AxisGraph) -> BTreeSet<(i32,i32)> {
    let c = coords(g);
    (0..g.num_nodes()).filter(|&i| g.legal_mask[i]).map(|i| c[i]).collect()
}
fn edge_at(g: &AxisGraph, e: usize) -> (usize, usize) { (g.edge_index.src[e] as usize, g.edge_index.dst[e] as usize) }
// edge map keyed by (src coord, dst coord) -> (axis, signed_dist, src_player); dummy edges skipped
fn edge_map(g: &AxisGraph) -> BTreeMap<((i32,i32),(i32,i32)), (usize, i32, i32)> {
    let c = coords(g);
    let mut m = BTreeMap::new();
    for e in 0..g.edge_index.src.len() {
        let (s, d) = edge_at(g, e);
        let a = &g.edge_attr.0[e*EDGE_FEAT_DIM..(e+1)*EDGE_FEAT_DIM];
        let axis = (0..3).find(|&k| a[k] == 1.0).unwrap_or(99);
        if axis == 99 { continue; }
        m.insert((c[s], c[d]), (axis, a[3] as i32, a[4] as i32));
    }
    m
}
fn spread(st: &[(i32,i32,i8)]) -> f64 {
    let n = st.len() as f64;
    let cq = st.iter().map(|s| s.0 as f64).sum::<f64>()/n;
    let cr = st.iter().map(|s| s.1 as f64).sum::<f64>()/n;
    st.iter().map(|s| ((s.0 as f64-cq).abs()).max((s.1 as f64-cr).abs())).fold(0.0, f64::max).max(1.0)
}

#[test]
fn scratch_d6_structural_losslessness_all_12_at_radius_8() {
    let mut rng = Lcg(20260918);
    let n_pos = 300;
    let mut spread_changed = [0usize; 12];
    let mut center_noneq = [0usize; 12];
    let mut n_off_window_any = 0usize;
    let mut axis_perm_seen: [BTreeSet<(usize,usize,i32)>; 12] = Default::default();
    let mut total_edges = 0usize;
    for _ in 0..n_pos {
        let n_st = rng.range(1, 61) as usize;
        let mut set = BTreeSet::new();
        while set.len() < n_st { set.insert((rng.range(-14, 15) as i32, rng.range(-14, 15) as i32)); }
        let stones: Vec<(i32,i32,i8)> = set.iter().enumerate().map(|(i,&(q,r))| (q, r, if i % 2 == 0 {1} else {-1})).collect();
        let g0 = build_axis_graph(&StoneList { stones: stones.clone() }, &R8);
        let l0 = legal_set(&g0); let s0 = stone_set(&g0); let e0 = edge_map(&g0);
        total_edges += e0.len();
        let sp0 = spread(&stones);
        if g0.policy_scatter_index.0.iter().any(|&x| x < 0) { n_off_window_any += 1; }
        for s in 0..12 {
            let rot: Vec<(i32,i32,i8)> = stones.iter().map(|&(q,r,p)| { let (a,b) = rotate_axial(q,r,s); (a,b,p) }).collect();
            let g1 = build_axis_graph(&StoneList { stones: rot.clone() }, &R8);
            let s0r: BTreeSet<_> = s0.iter().map(|&(q,r,p)| { let (a,b)=rotate_axial(q,r,s); (a,b,p) }).collect();
            assert_eq!(stone_set(&g1), s0r, "sym {s}: stone node set is not the rotated stone set");
            let l0r: BTreeSet<_> = l0.iter().map(|&(q,r)| rotate_axial(q,r,s)).collect();
            assert_eq!(legal_set(&g1), l0r, "sym {s}: legal-node set is not the rotated legal set");
            assert_eq!(g1.num_nodes(), g0.num_nodes(), "sym {s}: node count");
            let e1 = edge_map(&g1);
            assert_eq!(e1.len(), e0.len(), "sym {s}: edge count");
            for (&(a,b), &(ax, sd, sp)) in &e0 {
                let key = (rotate_axial(a.0,a.1,s), rotate_axial(b.0,b.1,s));
                let &(ax1, sd1, sp1) = e1.get(&key).unwrap_or_else(|| panic!("sym {s}: edge {a:?}->{b:?} missing after rotation"));
                assert_eq!(sd.abs(), sd1.abs(), "sym {s}: |signed_dist| changed");
                assert_eq!(sp, sp1, "sym {s}: src_player changed");
                assert_eq!(hexd(a,b), hexd(key.0, key.1), "sym {s}: hex distance changed");
                axis_perm_seen[s].insert((ax, ax1, sd.signum()*sd1.signum()));
            }
            if (spread(&rot) - sp0).abs() > 1e-9 { spread_changed[s] += 1; }
            let wc = rotate_axial(g0.window_center.0, g0.window_center.1, s);
            if wc != g1.window_center { center_noneq[s] += 1; }
        }
    }
    eprintln!("SCRATCH-D6 positions={n_pos} radius=8 total_edges={total_edges}: stones/legal/edges EXACT for all 12 elements");
    eprintln!("SCRATCH-D6 positions with any off-window (trunk-19) legal node: {n_off_window_any}/{n_pos}");
    eprintln!("SCRATCH-D6 spread (norm_q/norm_r scale) changed per element: {spread_changed:?}");
    eprintln!("SCRATCH-D6 window_center not equivariant per element: {center_noneq:?}");
    for s in 0..12 { eprintln!("SCRATCH-D6 sym {s}: (axis_in, axis_out, sign_product) = {:?}", axis_perm_seen[s]); }
    let ge = build_axis_graph(&StoneList::default(), &R8);
    let le = legal_set(&ge);
    let mut empty_noninv = vec![];
    for s in 0..12 { let r: BTreeSet<_> = le.iter().map(|&(q,r)| rotate_axial(q,r,s)).collect(); if r != le { empty_noninv.push(s); } }
    eprintln!("SCRATCH-D6 EMPTY board: legal set has {} cells; elements under which it is NOT invariant: {empty_noninv:?}", le.len());
}
